        // Tab switching SPA router logic

        function navigateTo(pageId) {

            // Find all section pages

            const pages = document.querySelectorAll('.page-transition');

            pages.forEach(page => {

                page.classList.add('hidden');

                page.style.opacity = '0';

                page.style.transform = 'translateY(10px)';

            });


            // Reveal targeted section page view window

            const activePage = document.getElementById(`page-${pageId}`);

            if (activePage) {

                activePage.classList.remove('hidden');

                setTimeout(() => {

                    activePage.style.opacity = '1';

                    activePage.style.transform = 'translateY(0)';

                }, 50);

            }


            // Sync visual desktop menu tab indicators

            const tabs = document.querySelectorAll('.nav-tab');

            tabs.forEach(tab => tab.classList.remove('active-tab'));

            

            const activeTab = document.getElementById(`tab-${pageId}`);

            if (activeTab) {

                activeTab.classList.add('active-tab');

            }


            // Hide mobile drop-down menu automatically after route select event completes

            document.getElementById('mobile-menu').classList.add('hidden');

            window.scrollTo({ top: 0, behavior: 'smooth' });

        }


        // Mobile toggle state controller

        const mobileMenuBtn = document.getElementById('mobile-menu-btn');

        const mobileMenu = document.getElementById('mobile-menu');


        mobileMenuBtn.addEventListener('click', () => {

            mobileMenu.classList.toggle('hidden');

        });
